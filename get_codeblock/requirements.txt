# get_codeblock — Python dependencies
#
# Python (.py) and Markdown (.md) work with ZERO third-party packages.
# The packages below add tree-sitter powered navigation for the brace/tagged
# languages. Install them into the SAME interpreter you run get_codeblock with:
#
#     <python> -m pip install -r requirements.txt
#
# get_codeblock checks these automatically: if a language you use needs a package
# that isn't installed, it prints the exact install command for your interpreter
# (instead of a traceback). Python/Markdown need nothing from this list.

tree-sitter>=0.23        # core parsing engine (needed by every grammar below)
tree-sitter-cpp>=0.23    # C / C++   (.cpp .cc .cxx .c++ .hpp .hh .hxx .h .c)
tree-sitter-c-sharp>=0.23 # C#        (.cs)
tree-sitter-typescript>=0.23 # TypeScript / JavaScript (.ts .js)
